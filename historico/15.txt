6.1 Corrigir tratarRetorno (bug crítico)

O log do 6023 mostrou:
text

Retorno: cStat=104 xMotivo=Lote processado nProt=151260000373648
Status atualizado: 4 (104 Lote processado)

Mas o XML real era cStat=100. O tratarRetorno está lendo o cStat da raiz (retEnviNFe, que é 104) em vez do protNFe (que é 100).

Substitui o método tratarRetorno inteiro no NfceTransmissor.php:
php

private function tratarRetorno($xmlRetorno, $xmlAssinado) {
    $cStatLote   = $this->getTag('cStat',   $xmlRetorno);
    $xMotivoLote = $this->getTag('xMotivo', $xmlRetorno);

    $cStat   = $cStatLote;
    $xMotivo = $xMotivoLote;
    $nProt   = '';
    $digVal  = '';

    if (preg_match('/<protNFe[^>]*>(.*?)<\/protNFe>/s', $xmlRetorno, $m)) {
        $prot = $m[1];
        if (preg_match('/<cStat>(\d+)<\/cStat>/', $prot, $mm))       $cStat   = $mm[1];
        if (preg_match('/<xMotivo>([^<]*)<\/xMotivo>/', $prot, $mm)) $xMotivo = $mm[1];
        if (preg_match('/<nProt>(\d+)<\/nProt>/', $prot, $mm))       $nProt   = $mm[1];
        if (preg_match('/<digVal>([^<]*)<\/digVal>/', $prot, $mm))   $digVal  = $mm[1];
    }

    $this->log("Retorno: cStatLote=$cStatLote | cStat=$cStat | xMotivo=$xMotivo | nProt=$nProt");

    if ($cStat === self::CSTAT_AUTORIZADA) {
        $this->atualizarAutorizada($nProt, $digVal, $xmlRetorno);
        return [
            'sucesso'   => true,
            'status'    => self::STATUS_AUTORIZADA,
            'cStat'     => $cStat,
            'protocolo' => $nProt,
            'mensagem'  => $xMotivo,
        ];
    }

    if ($cStat === self::CSTAT_DUPLICIDADE) {
        $this->atualizarStatus(self::STATUS_REJEITADA, $cStat, $xMotivo);
        return [
            'sucesso'   => false,
            'status'    => self::STATUS_REJEITADA,
            'cStat'     => $cStat,
            'mensagem'  => "Duplicidade: $xMotivo",
        ];
    }

    $this->atualizarStatus(self::STATUS_REJEITADA, $cStat, $xMotivo);
    return [
        'sucesso'   => false,
        'status'    => self::STATUS_REJEITADA,
        'cStat'     => $cStat,
        'mensagem'  => "SEFAZ ($cStat): $xMotivo",
    ];
}

emitirNfce: async (venda, tipoOperacao = 'FACILIT') => {
  const response = await api.post('/app/nfceEmitir', {
    empresa: EMPRESA_PADRAO,
    venda,
    tipo_operacao: tipoOperacao,
  });
  return response.data;

finalizarEEmitir: async (params) => {
  const resultadoVenda = await pagamentoService.finalizarVenda(params);
  if (!resultadoVenda?.sucesso) throw new Error(resultadoVenda?.erro || 'Falha ao finalizar');

  let nfce = null;
  try {
    nfce = await pagamentoService.emitirNfce(params.venda, params.origem ?? 'FACILIT');
  } catch (err) {
    nfce = { sucesso: false, erro: err.message };
  }
  return { venda: resultadoVenda, nfce };
},
