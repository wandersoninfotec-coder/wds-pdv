7.1 Auto-reprocessamento (próximo)

Adicionar useEffect no AlertaContingencia.jsx:
javascript

const [autoReprocessando, setAutoReprocessando] = useState(false);

useEffect(() => {
  if (!status) return;
  if (!status.online) return;
  if (status.contingencia_pendentes === 0) return;
  if (autoReprocessando || reprocessando) return;

  setAutoReprocessando(true);
  pagamentoService.reprocessarContingencia(20)
    .then(() => consultar())
    .catch((err) => console.error('[Auto-contingência] Erro:', err))
    .finally(() => setAutoReprocessando(false));
}, [status, autoReprocessando, reprocessando, consultar]);
