private function blocoReformaTotal() {
    // IS primeiro (VB6 faz ftIS antes de tudo) — sempre gravado
    $sql = $this->db->prepare("SELECT COALESCE(SUM(vIS), 0) FROM dfe_is
                               WHERE empresa = :emp AND nfe = :num");
    $sql->bindValue(':emp', $this->empresa);
    $sql->bindValue(':num', $this->numeroNfce);
    $sql->execute();
    $vIS = (float)$sql->fetchColumn();
    $this->add('vIS_W33', $this->fmtDec($vIS));

    $sql = $this->db->prepare("SELECT * FROM dfe_gibstotal
                               WHERE empresa = :emp AND nfe = :num AND modelo = 65 LIMIT 1");
    $sql->bindValue(':emp', $this->empresa);
    $sql->bindValue(':num', $this->numeroNfce);
    $sql->execute();
    $ibs = $sql->fetch(PDO::FETCH_ASSOC);

    $sql = $this->db->prepare("SELECT * FROM dfe_gcbstotal
                               WHERE empresa = :emp AND nfe = :num AND modelo = 65 LIMIT 1");
    $sql->bindValue(':emp', $this->empresa);
    $sql->bindValue(':num', $this->numeroNfce);
    $sql->execute();
    $cbs = $sql->fetch(PDO::FETCH_ASSOC);

    if (!$ibs && !$cbs) return;

    $sql = $this->db->prepare("SELECT COALESCE(SUM(bc_ibscbs), 0) FROM dfe_gibsuf
                               WHERE empresa = :emp AND nfe = :num");
    $sql->bindValue(':emp', $this->empresa);
    $sql->bindValue(':num', $this->numeroNfce);
    $sql->execute();
    $bcTotal = (float)$sql->fetchColumn();

    $this->add('vBCIBSCBS_W35', $this->fmtDec($bcTotal));

    if ($ibs) {
        $this->add('vDif_W38',            $this->fmtDec($ibs['vDif_UF']));
        $this->add('vDif_W43',            $this->fmtDec($ibs['vDif_Mun']));
        $this->add('vDevTrib_W39',        $this->fmtDec($ibs['vDevTrib_UF']));
        $this->add('vIBSUF_W41',          $this->fmtDec($ibs['vIBSUF_UF']));
        $this->add('vDevTrib_W44',        $this->fmtDec($ibs['vDevTrib_Mun']));
        $this->add('vIBSMun_W46',         $this->fmtDec($ibs['vIBS_Mun']));
        $this->add('vIBS_W47',            $this->fmtDec($ibs['vIBS']));
        $this->add('vCredPres_W48',       $this->fmtDec($ibs['vCredPres']));
        $this->add('vCredPresCondSus_W49',$this->fmtDec($ibs['vCredPresCondSus']));
    }
    if ($cbs) {
        $this->add('vDif_W53',            $this->fmtDec($cbs['vDif']));
        $this->add('vDevTrib_W54',        $this->fmtDec($cbs['vDevTrib']));
        $this->add('vCBS_W56',            $this->fmtDec($cbs['vCBS']));
        $this->add('vCredPres_W56a',      $this->fmtDec($cbs['vCredPres']));
        $this->add('vCredPresCondSus_W56b',$this->fmtDec($cbs['vCredPresCondSus']));
    }
}
